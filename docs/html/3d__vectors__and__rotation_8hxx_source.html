135 inline Eigen::Matrix<myfloat,3,3> skew(Eigen::Matrix<myfloat,3,1> v){
136  Eigen::Matrix<myfloat,3,3> E;
137  E << 0, -v(2), v(1),
138  v(2), 0, -v(0),
139  -v(1), v(0), 0;
140  return E;
141 }

143 inline Eigen::Matrix<myfloat,3,1> unskew(Eigen::Matrix<myfloat,3,3> E){
144  Eigen::Matrix<myfloat,3,1> v;
145  v << E(2,1), E(0,2), E(1,0);
