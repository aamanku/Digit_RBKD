117  std::pair<std::vector<std::pair<int, int>>, std::vector<std::pair<int, int>>>
118  get_casadi_function_io_dims(const Function &f, bool print = false) {
119  const casadi_int num_args = f.sz_arg();
120  const casadi_int num_res = f.sz_res();
121  std::vector<std::pair<int, int>> in_dims;
122  std::vector<std::pair<int, int>> res_dims;
123  for (casadi_int i = 0; i < num_args; ++i) {
124  in_dims.push_back({f.sparsity_in(i).rows(), f.sparsity_in(i).columns()});
125  }
126  for (casadi_int i = 0; i < num_res; ++i) {
127  res_dims.push_back({f.sparsity_out(i).rows(), f.sparsity_out(i).columns()});
128  }
129  if (print) {
130  std::cout << "Input dimensions: " << std::endl;
131  for (auto &in_dim: in_dims) {
132  std::cout << in_dim.first << "x" << in_dim.second << std::endl;
133  }
134  std::cout << "Output dimensions: " << std::endl;
135  for (auto &res_dim: res_dims) {
136  std::cout << res_dim.first << "x" << res_dim.second << std::endl;
137  }
138  }
139  return {in_dims, res_dims};
140  }

180  std::string generate_eigen_function_signature(const std::vector<std::pair<int, int>> &argument_dims,
181  const std::vector<std::pair<int, int>> &result_dims,
182  const std::vector<std::string> &argument_names,
183  const std::vector<std::string> &result_names,
184  const std::string &function_name) {
185  // create a void function signature
186  // with arguments as references to Eigen matrices with appropriate dimensions
187 
188  std::string function = "void " + function_name + "_eigen(";
189  for (int i = 0; i < argument_dims.size(); ++i) {
190  if (i != 0) {
191  function += "\t\t\t\t";
192  }
193  function += "const Eigen::Matrix<casadi_real," + std::to_string(argument_dims[i].first) + "," +
194  std::to_string(argument_dims[i].second) + "> &" + argument_names[i];
195  function += ",\n";
196  }
197  for (int i = 0; i < result_dims.size(); ++i) {
198  function += "\t\t\t\tEigen::Matrix<casadi_real," + std::to_string(result_dims[i].first) + "," +
199  std::to_string(result_dims[i].second) + "> &" + result_names[i];
200  if (i < result_dims.size() - 1) {
201  function += ", \n";
202  }
203  }
204  function += ")";
205 
206  return function;
207 
208  }

218  std::string generate_eigen_function_source(const std::vector<std::pair<int, int>> &argument_dims,
219  const std::vector<std::pair<int, int>> &result_dims,
220  const std::vector<std::string> &argument_names,
221  const std::vector<std::string> &result_names,
222  const std::string &function_name) {
223  // create a void function implementation
224  std::string source = "extern \"C\" CASADI_SYMBOL_EXPORT ";
225  source += generate_eigen_function_signature(argument_dims,
226  result_dims,
227  argument_names,
228  result_names,
229  function_name);
230  source += " {\n";
231  source += "// create arguments for generated casadi function\n";
232  source += "const casadi_real *arg[] = {";
233  for (int i = 0; i < argument_names.size(); ++i) {
234  source += argument_names[i] + ".data()";
235  if (i < argument_names.size() - 1) {
236  source += ", ";
237  }
238  }
239  source += "};\n";
240  source += "// create results for generated casadi function\n";
241  source += "casadi_real *res[] = {";
242  for (int i = 0; i < result_names.size(); ++i) {
243  source += result_names[i] + ".data()";
244  if (i < result_names.size() - 1) {
245  source += ", ";
246  }
247  }
248  source += "};\n";
249  source += "// call the generated casadi function\n";
250  source += function_name + "(arg, res, nullptr, nullptr, 0);\n";
251  source += "}\n";
252  return source;
