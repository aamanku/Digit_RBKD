44  inline Eigen::Matrix<myfloat,6,6> crm(Eigen::Matrix<myfloat,6,1> v) {
45  Eigen::Matrix<myfloat,6,6> vcross;
46  vcross << 0, -v(2), v(1), 0, 0, 0,
47  v(2), 0, -v(0), 0, 0, 0,
48  -v(1), v(0), 0, 0, 0, 0,
49  0, -v(5), v(4), 0, -v(2), v(1),
50  v(5), 0, -v(3), v(2), 0, -v(0),
51  -v(4), v(3), 0, -v(1), v(0), 0;
52  return vcross;
53  }

85  inline Eigen::Matrix<myfloat,6,6> mcI_to_rbi(myfloat m, Eigen::Matrix<myfloat,3,1> c, Eigen::Matrix<myfloat,3,3> I) {
86  Eigen::Matrix<myfloat,6,6> rbi;
87  rbi.block<3,3>(0,0) = I + m*crm(c)*crm(c).transpose();
88  rbi.block<3,3>(0,3) = m*crm(c);
89  rbi.block<3,3>(3,0) = m*crm(c).transpose();
90  rbi.block<3,3>(3,3) = m*Eigen::Matrix<myfloat,3,3>::Identity();
91  return rbi;
92  }

111  inline mcI_struct rbi_to_mcI(Eigen::Matrix<myfloat,6,6> rbi) {
112  mcI_struct mcI;
113  mcI.m = rbi(5,5);
114  Eigen::Matrix<myfloat,3,3> mC = rbi.block<3,3>(0,3);
115  mcI.c = unskew(mC)/mcI.m;
116  mcI.I = rbi.block<3,3>(0,0) - mC*mC.transpose()/mcI.m;
117  return mcI;
